void SetupScreen::renderTabRig(SDL_Renderer *renderer, int /*cx*/, int pad,
                               int fieldW, int fieldH, int fieldX,
                               int textPad) {
  auto *cat = fontMgr_.catalog();
  int y =
      (modalRect_.y + cat->ptSize(FontStyle::MediumBold) + 2 * pad + fieldH);
  int vSpace = pad / 2;
  SDL_Color white = {255, 255, 255, 255};
  SDL_Color orange = {255, 165, 0, 255};
  SDL_Color gray = {140, 140, 140, 255};

  // --- Rig Section ---
  cat->drawText(renderer, "Rig / CAT Control:", fieldX, y, white,
                FontStyle::SmallBold);
  y += cat->ptSize(FontStyle::SmallBold) + pad / 2;

  int rigLabelH = cat->ptSize(FontStyle::SmallRegular) + 4;
  cat->drawText(renderer, "rigctld Host (IP or Name):", fieldX, y, white,
                FontStyle::SmallRegular);
  rigHostRect_ = {fieldX, y, fieldW, rigLabelH + fieldH};
  y += rigLabelH;
  rigHostInput_.render(renderer, fontMgr_, fieldX, y, fieldW, fieldH,
                       FontStyle::SmallRegular, textPad, activeField_ == 0,
                       true, orange, gray, white, white, gray,
                       "e.g. localhost");
  y += fieldH + vSpace;

  cat->drawText(renderer, "rigctld Port:", fieldX, y, white,
                FontStyle::SmallRegular);
  rigPortRect_ = {fieldX, y, fieldW, rigLabelH + fieldH};
  y += rigLabelH;
  rigPortInput_.render(renderer, fontMgr_, fieldX, y, fieldW, fieldH,
                       FontStyle::SmallRegular, textPad, activeField_ == 1,
                       true, orange, gray, white, white, gray, "4532");
  y += fieldH + vSpace;

  // Auto-tune Toggle
  {
    SDL_Rect r = {fieldX, y, 20, 20};
    toggleRect_ = r; // reuse generic toggleRect_ for onMouseUp (needs care if multiple toggles)
    // Actually we should have specific rects for each toggle.
    // I'll update the class to have them.
    
    SDL_SetRenderDrawColor(renderer, 50, 50, 60, 255);
    SDL_RenderFillRect(renderer, &r);
    SDL_SetRenderDrawColor(renderer, 100, 100, 120, 255);
    SDL_RenderDrawRect(renderer, &r);
    if (rigAutoTune_) {
      SDL_SetRenderDrawColor(renderer, 0, 255, 0, 255);
      SDL_Rect check = {r.x + 4, r.y + 4, 12, 12};
      SDL_RenderFillRect(renderer, &check);
    }
    cat->drawText(renderer, "Enable Auto-Tune on DX Spot click", fieldX + 30,
                  y + 10, white, FontStyle::SmallRegular, false, false, true);
    y += 24 + pad;
  }

  // --- Rotator Section ---
  cat->drawText(renderer, "Rotator Control:", fieldX, y, white,
                FontStyle::SmallBold);
  y += cat->ptSize(FontStyle::SmallBold) + pad / 2;

  cat->drawText(renderer, "rotctld Host (IP or Name):", fieldX, y, white,
                FontStyle::SmallRegular);
  rotatorHostRect_ = {fieldX, y, fieldW, rigLabelH + fieldH};
  y += rigLabelH;
  rotatorHostInput_.render(renderer, fontMgr_, fieldX, y, fieldW, fieldH,
                           FontStyle::SmallRegular, textPad, activeField_ == 2,
                           true, orange, gray, white, white, gray,
                           "e.g. localhost");
  y += fieldH + vSpace;

  cat->drawText(renderer, "rotctld Port:", fieldX, y, white,
                FontStyle::SmallRegular);
  rotatorPortRect_ = {fieldX, y, fieldW, rigLabelH + fieldH};
  y += rigLabelH;
  rotatorPortInput_.render(renderer, fontMgr_, fieldX, y, fieldW, fieldH,
                           FontStyle::SmallRegular, textPad, activeField_ == 3,
                           true, orange, gray, white, white, gray, "4533");
  y += fieldH + vSpace;

  // Auto-track and Upover Toggles
  {
    // Auto-track
    SDL_Rect r1 = {fieldX, y, 20, 20};
    rotatorAutoTrackRect_ = r1;
    SDL_SetRenderDrawColor(renderer, 50, 50, 60, 255);
    SDL_RenderFillRect(renderer, &r1);
    SDL_SetRenderDrawColor(renderer, 100, 100, 120, 255);
    SDL_RenderDrawRect(renderer, &r1);
    if (rotatorAutoTrack_) {
      SDL_SetRenderDrawColor(renderer, 0, 255, 0, 255);
      SDL_Rect check = {r1.x + 4, r1.y + 4, 12, 12};
      SDL_RenderFillRect(renderer, &check);
    }
    cat->drawText(renderer, "Auto-track Satellite", fieldX + 30, y + 10, white,
                  FontStyle::SmallRegular, false, false, true);

    // Upover
    SDL_Rect r2 = {fieldX + fieldW / 2, y, 20, 20};
    rotatorUpoverRect_ = r2;
    SDL_SetRenderDrawColor(renderer, 50, 50, 60, 255);
    SDL_RenderFillRect(renderer, &r2);
    SDL_SetRenderDrawColor(renderer, 100, 100, 120, 255);
    SDL_RenderDrawRect(renderer, &r2);
    if (rotatorUpover_) {
      SDL_SetRenderDrawColor(renderer, 0, 255, 0, 255);
      SDL_Rect check = {r2.x + 4, r2.y + 4, 12, 12};
      SDL_RenderFillRect(renderer, &check);
    }
    cat->drawText(renderer, "Enable Upover Mode (Az flip)", r2.x + 30, y + 10,
                  white, FontStyle::SmallRegular, false, false, true);
    
    y += 24 + pad;
  }

  cat->drawText(renderer, "Control requires 'rigctld'/'rotctld' (Hamlib) running.",
                fieldX, y, gray, FontStyle::Fast);
}
